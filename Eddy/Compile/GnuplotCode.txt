procedure GnuplotInstall;
var
  ResultCode: Integer;
begin
  if not Exec(ExpandConstant('{#GnuplotInstallDir}\gp526-win64-mingw_2.exe'), '', '', SW_SHOWNORMAL,
    ewWaitUntilTerminated, ResultCode)
  then
    MsgBox('Gnuplot failed to install!' + #13#10 +
      SysErrorMessage(ResultCode), mbError, MB_OK);
end;
